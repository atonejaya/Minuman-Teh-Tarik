### Task 8: Stok Sales Request → Admin Approve + Notifikasi (#4)

**Migration:**

```sql
CREATE TABLE IF NOT EXISTS "StockRequest" (
  id BIGSERIAL PRIMARY KEY,
  request_number TEXT UNIQUE NOT NULL,
  sales_id BIGINT REFERENCES "User"(id),
  request_date DATE NOT NULL DEFAULT CURRENT_DATE,
  status TEXT NOT NULL DEFAULT 'PENDING' CHECK (status IN ('PENDING', 'APPROVED', 'REJECTED')),
  total_qty INT DEFAULT 0,
  notes TEXT,
  approved_by BIGINT REFERENCES "User"(id),
  approved_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS "StockRequestItem" (
  id BIGSERIAL PRIMARY KEY,
  request_id BIGINT REFERENCES "StockRequest"(id) ON DELETE CASCADE,
  product_id BIGINT REFERENCES "Product"(id),
  qty INT NOT NULL DEFAULT 1,
  unit_id BIGINT REFERENCES "Unit"(id),
  remark TEXT
);

-- RPC Functions
CREATE OR REPLACE FUNCTION public.stock_request_submit(p_sales_id BIGINT, p_items JSONB, p_notes TEXT DEFAULT NULL)
RETURNS BIGINT AS $$
DECLARE
  v_request_id BIGINT;
  v_number TEXT;
  v_total INT := 0;
  v_item JSONB;
BEGIN
  SELECT 'SR-' || TO_CHAR(NOW(), 'YYYYMMDD') || '-' || LPAD(COALESCE(MAX(CAST(SUBSTRING(request_number FROM 12) AS INT)), 0) + 1, 4, '0')
  INTO v_number FROM public."StockRequest" WHERE request_number LIKE 'SR-' || TO_CHAR(NOW(), 'YYYYMMDD') || '-%';

  INSERT INTO public."StockRequest" (request_number, sales_id, request_date, status, notes)
  VALUES (v_number, p_sales_id, CURRENT_DATE, 'PENDING', p_notes)
  RETURNING id INTO v_request_id;

  FOR v_item IN SELECT * FROM jsonb_array_elements(p_items)
  LOOP
    v_total := v_total + (v_item->>'qty')::INT;
    INSERT INTO public."StockRequestItem" (request_id, product_id, qty, unit_id, remark)
    VALUES (v_request_id, (v_item->>'product_id')::BIGINT, (v_item->>'qty')::INT,
            (v_item->>'unit_id')::BIGINT, v_item->>'remark');
  END LOOP;

  UPDATE public."StockRequest" SET total_qty = v_total WHERE id = v_request_id;
  RETURN v_request_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE FUNCTION public.stock_request_approve(p_request_id BIGINT, p_approved_by BIGINT)
RETURNS VOID AS $$
BEGIN
  UPDATE public."StockRequest" SET status = 'APPROVED', approved_by = p_approved_by, approved_at = NOW()
  WHERE id = p_request_id AND status = 'PENDING';
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE FUNCTION public.stock_request_reject(p_request_id BIGINT, p_approved_by BIGINT, p_reason TEXT DEFAULT NULL)
RETURNS VOID AS $$
BEGIN
  UPDATE public."StockRequest" SET status = 'REJECTED', approved_by = p_approved_by, approved_at = NOW(), notes = COALESCE(p_reason, notes)
  WHERE id = p_request_id AND status = 'PENDING';
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```

```jsx
import React, { useState, useEffect } from 'react';
import { MapContainer, TileLayer, Marker, Popup, Polyline } from 'react-leaflet';
import { supabase } from '../../../utils/supabase';
import 'leaflet/dist/leaflet.css';

const RouteMapPage = () => {
  const [warungs, setWarungs] = useState([]);
  const [visits, setVisits] = useState([]);
  const [selectedSales, setSelectedSales] = useState(null);

  useEffect(() => {
    const fetchData = async () => {
      // Fetch warung dengan visit_order
      const { data: w } = await supabase.from('Warung')
        .select('id, name, address, latitude, longitude, visit_order, assigned_sales_id')
        .eq('status', 'ACTIVE')
        .not('latitude', 'is', null);
      setWarungs(w || []);

      // Fetch visit hari ini
      const today = new Date().toISOString().slice(0, 10);
      const { data: v } = await supabase.from('SalesVisit')
        .select('id, warung_id, sales_id, status, check_in_time, check_out_time')
        .eq('visit_date', today);
      setVisits(v || []);
    };
    fetchData();
  }, []);

  // Filter by selected sales
  const filteredWarungs = selectedSales
    ? warungs.filter(w => w.assigned_sales_id === selectedSales)
    : warungs;

  // Sort by visit_order
  const sortedWarungs = [...filteredWarungs].sort((a, b) => (a.visit_order || 0) - (b.visit_order || 0));

  // Create route polyline
  const routePoints = sortedWarungs
    .filter(w => w.latitude && w.longitude)
    .map(w => [Number(w.latitude), Number(w.longitude)]);

  // Get visit status for each warung
  const getVisitStatus = (warungId) => {
    const visit = visits.find(v => v.warung_id === warungId);
    return visit ? visit.status : null;
  };

  return (
    <div style={{ height: 'calc(100vh - 64px)' }}>
      {/* Sales filter */}
      <div style={{ padding: '12px', background: 'var(--surface)' }}>
        <select onChange={(e) => setSelectedSales(Number(e.target.value) || null)}>
          <option value="">Semua Sales</option>
          {/* ... sales options */}
        </select>
      </div>

      {/* Map */}
      <MapContainer center={[-6.2088, 106.8456]} zoom={12} style={{ height: '100%' }}>
        <TileLayer url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png" />
        {routePoints.length > 1 && <Polyline positions={routePoints} color="var(--primary)" />}
        {sortedWarungs.map((w, i) => {
          const status = getVisitStatus(w.id);
          return (
            <Marker key={w.id} position={[Number(w.latitude), Number(w.longitude)]}>
              <Popup>
                <strong>{i + 1}. {w.name}</strong><br />
                {w.address}<br />
                Status: {status || 'Belum dikunjungi'}
              </Popup>
            </Marker>
          );
        })}
      </MapContainer>
    </div>
  );
};
export default RouteMapPage;
```
